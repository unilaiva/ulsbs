% Loaded by ulsbs-book-defaults.code.tex after ulsbs-init.code.tex and utilities,

% Owns permission/request state, effective predicates, local dispatch, short
% switches and musical guards. Effective visibility is permission AND request;
% requests obey ordinary grouping and never grant book permission. Permissions
% freeze globally at begindocument/before; local requests remain scope-local.

  {\legacy_if:nT {ulShowMusicComments} {#1}}

  {
    \bool_if:nTF
      {\legacy_if_p:n {ulShowChords} || \legacy_if_p:n {ulShowNotes}
        || \legacy_if_p:n {ulShowLilypond}}
      {#1} {#2}
