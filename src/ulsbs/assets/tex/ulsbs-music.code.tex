% #1 is token source, #2 an output tl variable. Walk actual TeX tokens, preserving
% spaces and copying brace groups opaquely (no recursive transposition within
% authored arguments). Unknown macros stay unexpanded: this is not another source
% evaluation. Export only the result across the scratch group's end.
\cs_new_protected:Npn \__ulsbs_music_transpose:nN #1#2
  {\group_begin:
   \tl_clear:N \l__ulsbs_music_result_tl
   \tl_clear:N \l__ulsbs_music_pending_tl
   \__ulsbs_music_prepare:
   \__ulsbs_music_scan: #1\q__ulsbs_music_end
   \exp_args:NNNV \group_end: \tl_set:Nn #2 \l__ulsbs_music_result_tl}

% #1 is an interactive offset, #2 effective-original tokens, #3 output tl. Set
% both layer and interval locally so key preparation moves the destination, not
% the inferred source; ambient source offsets/preferences survive this operation.
\cs_new_protected:Npn \__ulsbs_music_transpose_alternative:nnN #1#2#3
  {\group_begin:
   \int_set:Nn \l__ulsbs_music_layer_int {#1}
   \int_set:Nn \l__ulsbs_music_transpose_int {#1}
   \__ulsbs_music_transpose:nN {#2}\l__ulsbs_music_effective_tl
   \exp_args:NNNV \group_end: \tl_set:Nn #3 \l__ulsbs_music_effective_tl}

% Sentinel terminates this token stream. Spaces/groups commit a pending A--G
% first; groups are copied whole, keeping command arguments and scope intact.
\cs_new_protected:Npn \__ulsbs_music_scan:
  {\peek_meaning_remove:NTF \q__ulsbs_music_end
    {\__ulsbs_music_commit:}
    {\peek_meaning_remove:NTF \c_space_token
      {\__ulsbs_music_commit:\tl_put_right:Nn \l__ulsbs_music_result_tl {~}
       \__ulsbs_music_scan:}
      {\peek_catcode:NTF \c_group_begin_token
        {\__ulsbs_music_group:n}{\__ulsbs_music_token:N}}}}

% #1 is group content, restored with braces after committing the preceding pitch.
\cs_new_protected:Npn \__ulsbs_music_group:n #1
  {\__ulsbs_music_commit:
   \tl_put_right:Nn \l__ulsbs_music_result_tl {{#1}}
   \__ulsbs_music_scan:}

% #1 is one unexpanded token. notrans owns the next argument, which must bypass
% pitch recognition as a unit rather than leaving its content in this scan.
\cs_new_protected:Npn \__ulsbs_music_token:N #1
  {\token_if_eq_meaning:NNTF #1\notrans
    {\__ulsbs_music_commit:\__ulsbs_music_literal_argument:Nn #1}
    {\__ulsbs_music_token_regular:N #1}}

% #1 is notrans, #2 its next argument; preserve both without evaluation, then

  {\tl_put_right:Nn \l__ulsbs_music_result_tl {#1{#2}}\__ulsbs_music_scan:}

% Recognize literal uppercase pitch letters and accidental shorthand/macros.
% Every other token commits the pending pitch then passes through unchanged.
\cs_new_protected:Npn \__ulsbs_music_token_regular:N #1
  {\str_case:enF {\token_to_str:N #1}
    {{A}{\__ulsbs_music_pending:N #1}{B}{\__ulsbs_music_pending:N #1}
     {C}{\__ulsbs_music_pending:N #1}{D}{\__ulsbs_music_pending:N #1}
     {E}{\__ulsbs_music_pending:N #1}{F}{\__ulsbs_music_pending:N #1}
     {G}{\__ulsbs_music_pending:N #1}}
    {\str_if_eq:eeTF {\token_to_str:N #1}{\c_hash_str}
      {\__ulsbs_music_sign:n {1}}
      {\str_if_eq:eeTF {\token_to_str:N #1}{&}
        {\__ulsbs_music_sign:n {-1}}
        {\token_if_eq_meaning:NNTF #1\shrp
          {\__ulsbs_music_sign:n {1}}
          {\token_if_eq_meaning:NNTF #1\flt
            {\__ulsbs_music_sign:n {-1}}
            {\__ulsbs_music_commit:\tl_put_right:Nn \l__ulsbs_music_result_tl {#1}}}}}}
   \__ulsbs_music_scan:}

% Convert a pending letter/accidental to pitch, spell only for nonzero offsets,
% append protected glyph tokens, then clear the pending letter. Zero-offset
% notation is not enharmonically respelled merely because layers are enabled.
\cs_new_protected:Npn \__ulsbs_music_commit:
  {\tl_if_empty:NF \l__ulsbs_music_pending_tl
    {\int_set:Nn \l__ulsbs_music_letter_int
      {\str_case:Vn \l__ulsbs_music_pending_tl {{C}{0}{D}{1}{E}{2}{F}{3}{G}{4}{A}{5}{B}{6}}}
     \int_set:Nn \l__ulsbs_music_pitch_int
      {\int_mod:nn {1200+\clist_item:Nn \c__ulsbs_pdf_natural_clist
        {\l__ulsbs_music_letter_int+1}+\l__ulsbs_music_accidental_int
        +\l__ulsbs_music_transpose_int}{12}}
     \int_if_zero:nF {\l__ulsbs_music_transpose_int}
      {\bool_if:NTF \l__ulsbs_music_key_bool
        {\__ulsbs_music_spell_key:}{\__ulsbs_music_spell_pitch:}}
     \tl_put_right:Nx \l__ulsbs_music_result_tl
      {\clist_item:Nn \c__ulsbs_pdf_letters_clist {\l__ulsbs_music_letter_int+1}
       \int_case:nn {\l__ulsbs_music_accidental_int}
        {{-1}{\exp_not:N \flt}{1}{\exp_not:N \shrp}}}
     \tl_clear:N \l__ulsbs_music_pending_tl}}
