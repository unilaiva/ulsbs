% No arguments. Freeze label/target/selection and open the song-local group,
% outer song box and body box. Hooks and index staging execute now, not during
% later layout probes; prefix/anchor nodes are retained for final assembly.
\cs_new_protected:Npn \__ulsbs_song_start:
  {
    \bool_if:NT \g__ulsbs_song_heading_pending_bool
    {\bool_gset_false:N \g__ulsbs_song_heading_pending_bool
     \if@nobreak\ifdim\pagetotal>0pt
      % A standalone heading consumes capacity in the first song column even
      % when the column itself moves to the next physical page.
      \dim_set:Nn \l__ulsbs_song_heading_reserve_dim {\pagetotal}
      \__ulsbs_layout_clear_page:\enlargethispage{-\l__ulsbs_song_heading_reserve_dim}
     \fi\fi}
    \char_set_active_eq:nN {35}\shrp
    \char_set_active_eq:nN {38}\flt
    \tl_set:Nx \l__ulsbs_song_label_tl {\theulSongNumber}
    \tl_set:Nx \l__ulsbs_song_target_tl
      {song\int_use:N \g__ulsbs_songs_serial_int-\theulSongNumber}
    \bool_set_true:N \l__ulsbs_song_selected_bool
    \bool_if:NT \g__ulsbs_song_selection_bool
      {\exp_args:NNV \clist_if_in:NnF \g__ulsbs_song_selection_clist \l__ulsbs_song_label_tl
        {\bool_set_false:N \l__ulsbs_song_selected_bool}}
    \protected@edef\@currentlabel{\p@ulSongNumber\theulSongNumber}
    \int_gincr:N \g__ulsbs_song_serial_int
    \tl_set:Nx \l__ulsbs_song_box_tl {g__ulsbs_song_\int_use:N \g__ulsbs_song_serial_int _box}
    \group_begin:
    \vbox_set:Nw \l__ulsbs_song_box
      \__ulsbs_layout_song_begin:
      \UseHook{ulsbs/song/begin}
      \ulResetTitles
       \__ulsbs_song_multikey:
       \__ulsbs_song_anchor:
       \vbox_set:Nw \l__ulsbs_song_prefix_box
       \ulSongTarget{\int_if_zero:nTF {\value{section}}{1}{2}}{\l__ulsbs_song_target_tl}
      \seq_map_inline:Nn \l__ulsbs_song_extra_indices_seq {##1}
      \seq_map_inline:Nn \l__ulsbs_song_titles_seq
        {\__ulsbs_song_record:nnn {title}{}{##1}}
       \tl_if_empty:NF \ulTagList
         {\__ulsbs_song_record:nnn {tag}{}{\ulTagList}}
       \__ulsbs_song_record:nnn {author}{}{\ulSongAuthors}
       \vbox_set_end:
       % Match legacy font first-use order without replaying body source.
       \vbox_set:Nw \l__ulsbs_song_body_box
        \nointerlineskip\ulLyricFontStyle\relax\__ulsbs_verse_baseline:
      \hypersetup{bookmarksdepth=0}\phantomsection
       % Choose now: neither shipout nor next-pass TOC reading may query later
       % local visibility. addcontentsline still retains the write in this box.
       \legacy_if:nTF {ulShowPhaseInToc}
         {\addcontentsline{toc}{subsection}{\numberline{\theulSongNumber}\ulSongTitle\relax
           \hspace{0.15em}\scriptsize\textsuperscript{\rmfamily\tiny\ulPhaseList}}}
         {\addcontentsline{toc}{subsection}{\numberline{\theulSongNumber}\ulSongTitle\relax}}
      \hypersetup{bookmarksdepth=2}
  }

% Stage semantic records once; only the retained box resolves page labels.
% #1: kind; #2: optional overriding index-ID list; #3: entry tokens. Filter by
% declared kind, run record hooks and stage nonempty entries when files/indices
% are enabled. Overrides and expansion scratch are confined to a local group.
\cs_new_protected:Npn \__ulsbs_song_record:nnn #1#2#3
  {
    \__ulsbs_song_require:
    \group_begin:
    \clist_if_empty:nF {#2}{\clist_set:Nn \l__ulsbs_song_indices_clist {#2}}
    \protected@edef\l_tmpa_tl{#3}
    \tl_if_empty:NF \l_tmpa_tl
      {
        \clist_map_inline:Nn \l__ulsbs_song_indices_clist
          {
            \prop_get:NnNTF \g__ulsbs_index_kinds_prop {##1}\l_tmpb_tl
              {
                \str_if_eq:VnT \l_tmpb_tl {#1}
                  {
                    \UseHook{ulsbs/index/record}
                    \prop_get:NnNTF \l__ulsbs_song_index_labels_prop {##1}\l_tmpb_tl
                      {}
                      {\tl_set_eq:NN \l_tmpb_tl \l__ulsbs_song_label_tl}
                    \if@filesw\ifulSongIndexes
                      \ulsbs_index_record:nnnnn {##1}{#3}{\l_tmpb_tl}
                        {\l__ulsbs_song_target_tl.\int_if_zero:nTF {\value{section}}{1}{2}}
                        {\thechapter}
                    \fi\fi
                  }
              }
              {\msg_fatal:nnn {ulsbs-song}{index}{##1}}
          }
      }
    \group_end:
  }
