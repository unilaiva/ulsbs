% Request a snippet's source key and twelve target tonics, or unknown-key nulls.
\cs_new_protected:Npn \__ulsbs_score_request:n #1
  {
    \__ulsbs_pdf_key_load:
    \tl_set:Nn \l__ulsbs_score_signature_tl {unknown}
    \tl_set:Nn \l__ulsbs_score_key_lua_tl {nil}
    \tl_set:Nn \l__ulsbs_score_targets_lua_tl {nil}
    \bool_if:NT \l__ulsbs_pdf_key_available_bool
      {
        \tl_set:Nx \l__ulsbs_score_signature_tl
          {\int_use:N\l__ulsbs_pdf_key_letter_int-
           \int_use:N\l__ulsbs_pdf_key_pitch_int-\int_use:N\l__ulsbs_pdf_key_minor_int}
        \tl_set:Nx \l__ulsbs_score_key_lua_tl
          {\c_left_brace_str letter=\int_use:N\l__ulsbs_pdf_key_letter_int,
           pitch=\int_use:N\l__ulsbs_pdf_key_pitch_int,
           minor=\int_use:N\l__ulsbs_pdf_key_minor_int\c_right_brace_str}
        \tl_set:Nx \l__ulsbs_score_targets_lua_tl {\c_left_brace_str}
        \int_step_inline:nnn {0}{11}
          {
            \int_set:Nn \l__ulsbs_pdf_target_pitch_int
              {\int_mod:nn {\l__ulsbs_pdf_key_pitch_int+##1}{12}}
            \int_if_zero:nTF {##1}
              {\int_set_eq:NN \l__ulsbs_pdf_target_letter_int \l__ulsbs_pdf_key_letter_int}
              {\__ulsbs_pdf_canonical_tonic:n {\l__ulsbs_pdf_target_pitch_int}}
            \tl_put_right:Nx \l__ulsbs_score_targets_lua_tl
              {\c_left_brace_str letter=\int_use:N\l__ulsbs_pdf_target_letter_int,
               pitch=\int_use:N\l__ulsbs_pdf_target_pitch_int\c_right_brace_str,}
          }
        \tl_put_right:Nx \l__ulsbs_score_targets_lua_tl {\c_right_brace_str}
      }
    \directlua{ulsbs_score_layers.request(
      "\luaescapestring{#1}","\luaescapestring{\ulsbs@pdfnameprefix}",
      "\luaescapestring{\ulSongId}","\luaescapestring{\tl_to_str:V\ulSongTitle}",
      \l__ulsbs_score_key_lua_tl,\l__ulsbs_score_targets_lua_tl)}
  }
% Overlay one snippet/system/offset variant in the displayed original's slot.
\cs_new_protected:Npn \__ulsbs_score_alternative:nnn #1#2#3
  {
    \box_clear:N \l__ulsbs_score_alt_box
    \bool_if:NT \l__ulsbs_pdf_key_available_bool
      {
        \cs_if_exist:cT
          {ULScoreVariant:\ulsbs@pdfnameprefix :#1:\l__ulsbs_score_signature_tl :t\__ulsbs_pdf_pad:nn {2}{#3}}
          {
            \tl_set:Nx \l__ulsbs_score_filename_tl
              {\use:c {ULScoreVariant:\ulsbs@pdfnameprefix :#1:\l__ulsbs_score_signature_tl
                :t\__ulsbs_pdf_pad:nn {2}{#3}}-#2}
            \exp_args:Nx \IfFileExists {\l__ulsbs_score_filename_tl.pdf}
              {\hbox_set:Nn \l__ulsbs_score_alt_box
                {\includegraphics{\l__ulsbs_score_filename_tl}}}
              {\PackageWarning{ulsbs}{Missing~score~#1~system~#2~offset~#3;
                 ~original~notation~substituted}}
          }
      }
    \fp_set:Nn \l__ulsbs_score_scale_fp {1}
    \dim_compare:nNnTF {\box_wd:N\l__ulsbs_score_alt_box}>{0pt}
      {
        \dim_compare:nNnTF
          {\box_ht:N\l__ulsbs_score_alt_box+\box_dp:N\l__ulsbs_score_alt_box}>{0pt}
          {\fp_set:Nn \l__ulsbs_score_scale_fp
            {min(1,
              \dim_to_fp:n {\box_wd:N\l__ulsbs_score_base_box}/
                \dim_to_fp:n {\box_wd:N\l__ulsbs_score_alt_box},
              \dim_to_fp:n {\box_ht:N\l__ulsbs_score_base_box+\box_dp:N\l__ulsbs_score_base_box}/
                \dim_to_fp:n {\box_ht:N\l__ulsbs_score_alt_box+\box_dp:N\l__ulsbs_score_alt_box})}}
          {\box_clear:N \l__ulsbs_score_alt_box}
      }
      {\box_clear:N \l__ulsbs_score_alt_box}
    % PDF dimensions round to sp on conversion. At the exact policy boundary,
    % rounding both boxes can leave a deficit of up to one sp in either axis.
    % Test this policy against the natural original, as in engraving validation.
    % The display limit is separate and may shrink an accepted alternative further.
    \fp_compare:nNnT
      {max(
        \c__ulsbs_score_minimum_scale_fp*\dim_to_fp:n {\box_wd:N\l__ulsbs_score_alt_box}
          -\dim_to_fp:n {\box_wd:N\l__ulsbs_score_base_box},
        \c__ulsbs_score_minimum_scale_fp*\dim_to_fp:n
          {\box_ht:N\l__ulsbs_score_alt_box+\box_dp:N\l__ulsbs_score_alt_box}
          -\dim_to_fp:n {\box_ht:N\l__ulsbs_score_base_box+\box_dp:N\l__ulsbs_score_base_box})}
      >{\dim_to_fp:n {1sp}}
      {\PackageWarning{ulsbs}{Score~#1~system~#2~offset~#3~is~too~small;
         ~original~notation~substituted}
       \box_clear:N \l__ulsbs_score_alt_box}
    \box_if_empty:NTF \l__ulsbs_score_alt_box
      {\smash{\rlap{\__ulsbs_pdf_mark:nn {#3}{\box_use:N \l__ulsbs_score_display_box}}}}
      {
        % Place the matrix origin at the original's top left. Changing the
        % matrix at shipout then preserves this anchor and all TeX dimensions.
        \hbox_set:Nn \l__ulsbs_score_transform_box
          {\pdfextension save\relax
           \pdfextension setmatrix{1~0~0~1}
           \rlap{\raisebox{-\box_ht:N\l__ulsbs_score_alt_box}{\box_use:N \l__ulsbs_score_alt_box}}
           \pdfextension restore\relax}
        \directlua{ulsbs_score_layers.tag_alternative(
          \number\l__ulsbs_score_transform_box,
          "\luaescapestring{\ulsbs@pdfnameprefix/#1/\int_use:N\l__ulsbs_score_fragment_int/#3}",
          \fp_to_decimal:n {min(\l__ulsbs_score_scale_fp,
            \dim_to_fp:n {\box_wd:N\l__ulsbs_score_display_box}/
              \dim_to_fp:n {\box_wd:N\l__ulsbs_score_alt_box},
            \dim_to_fp:n {\box_ht:N\l__ulsbs_score_display_box+\box_dp:N\l__ulsbs_score_display_box}/
              \dim_to_fp:n {\box_ht:N\l__ulsbs_score_alt_box+\box_dp:N\l__ulsbs_score_alt_box})})}
        \smash{\rlap{\__ulsbs_pdf_mark:nn {#3}
          {\raisebox{\box_ht:N\l__ulsbs_score_display_box}{\box_use_drop:N \l__ulsbs_score_transform_box}}}}
      }
  }
% Request and layer a snippet/system/file while retaining the original box size.
\cs_new_protected:Npn \__ulsbs_score_system:nnn #1#2#3
  {
    \group_begin:
    \bool_set_true:N \l__ulsbs_score_graphics_raw_bool
    \__ulsbs_score_request:n {#1}
    \hbox_set:Nn \l__ulsbs_score_base_box {\includegraphics{#3}}
    \box_set_eq:NN \l__ulsbs_score_display_box \l__ulsbs_score_base_box
    \__ulsbs_score_fit_line:N \l__ulsbs_score_display_box
    \__ulsbs_pdf_ensure_layers:
    \hbox_set:Nn \l__ulsbs_score_result_box
      {
        % Draft passes only need the request and the original's box.
        \ifulsbs@pdfdraftpass\else
          \int_step_inline:nnnn {11}{-1}{1}{\__ulsbs_score_alternative:nnn {#1}{#2}{##1}}
        \fi
        \__ulsbs_pdf_mark:nn {0}{\box_use_drop:N \l__ulsbs_score_display_box}
      }
    \leavevmode\box_use_drop:N \l__ulsbs_score_result_box
    \group_end:
  }
