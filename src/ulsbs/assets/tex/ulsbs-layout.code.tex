% No arguments. Apply requested 1/2-column kernel mode; 0 leaves it alone.
% On the first two-column start, retain any full-width prefix nodes and restore
% them via a one-shot output hook, reducing the first page's column capacity.
\cs_new_protected:Npn \__ulsbs_layout_columns:
  {
    \int_compare:nNnT {\g__ulsbs_layout_columns_int}>{0}
    {
    \topskip=1sp\relax
    \int_compare:nNnTF {\g__ulsbs_layout_columns_int}={2}
      {\if@twocolumn\else
         \par
         \bool_if:NTF \g__ulsbs_layout_has_output_bool {\twocolumn}
           {\ifdim\pagetotal>0pt
              % Move the already-typeset prefix; never replay its source.
              \dim_set:Nn \l__ulsbs_layout_prefix_height_dim {\pagetotal}
              \l__ulsbs_layout_saved_output_toks=\output
              \output{\global\setbox\g__ulsbs_layout_prefix_box=\vbox{\unvbox255}}
              \penalty-10000\relax
              \output=\l__ulsbs_layout_saved_output_toks
              \twocolumn
              \global\@colht=\dimexpr\textheight-\l__ulsbs_layout_prefix_height_dim\relax
              \global\@colroom=\@colht\global\vsize=\@colht
              \AddToHookNext{cmd/@combinedblfloats/before}
                {\setbox\@outputbox=\vbox{\unvbox\g__ulsbs_layout_prefix_box
                   \kern0pt\nointerlineskip\box\@outputbox}
                 \box_gclear:N \g__ulsbs_layout_prefix_box}
            \else\twocolumn\fi}
       \fi}
      {\if@twocolumn\onecolumn\fi}
    \topskip=0pt\relax
    }
  }

% No arguments. Reconcile scoped settings with physical mode before source song
% entry/end; selection defers physical mode changes until retained-box commit.
\cs_new_protected:Npn \__ulsbs_layout_prepare:
  {
    \bool_set_false:N \l__ulsbs_layout_scope_restored_bool
    \bool_if:NF \g__ulsbs_song_selection_bool
      {
        \if@twocolumn
          \int_compare:nNnF {\g__ulsbs_layout_columns_int}={2}
            {\__ulsbs_layout_restore_scope:}
        \else
          \int_compare:nNnT {\g__ulsbs_layout_columns_int}={2}
            {\__ulsbs_layout_restore_scope:}
        \fi
        \__ulsbs_layout_columns:
      }
  }

% Desired song settings are scoped, but the kernel column flags are global.
% On scope exit retain pending contributions across the kernel mode change:
% clearpage would otherwise move them into the old mode's next column. The
% temporary capture can run more than once while TeX drains an overfull page.
\cs_new_protected:Npn \__ulsbs_layout_restore_scope:
  {
    \bool_set_true:N \l__ulsbs_layout_scope_restored_bool
    \par
    \box_gclear:N \g__ulsbs_layout_scoped_column_box
    \l__ulsbs_layout_saved_output_toks=\output
    \output{\global\setbox\g__ulsbs_layout_scoped_column_box=\vbox{
      \unvbox\g__ulsbs_layout_scoped_column_box\unvbox255}}
    \penalty-10000\relax
    \output=\l__ulsbs_layout_saved_output_toks
    \if@twocolumn\if@firstcolumn\else\null\__ulsbs_layout_clear_page:\fi\fi
    \cs_set_eq:NN \__ulsbs_layout_saved_clear: \clearpage
    \cs_set_eq:NN \clearpage \prg_do_nothing:
    \int_compare:nNnTF {\g__ulsbs_layout_columns_int}={2}{\twocolumn}{\onecolumn}
    \cs_set_eq:NN \clearpage \__ulsbs_layout_saved_clear:
    \vbox_unpack_drop:N \g__ulsbs_layout_scoped_column_box
  }

% At an unchosen breakpoint the two skips cancel. At a chosen breakpoint
% TeX discards the compensating skip, leaving bottom glue on the outgoing page.
% #1: break penalty; emit a candidate boundary, preserving prior inner skip.
\cs_new_protected:Npn \__ulsbs_layout_break:n #1
  {
    \par
    \skip_zero:N \l__ulsbs_layout_previous_skip
    \ifinner
      \skip_set:Nn \l__ulsbs_layout_previous_skip {\lastskip}\unskip
    \fi
    \skip_set:Nn \l__ulsbs_layout_bottom_skip {\ulColumnBottomGlue}
    \nobreak
    \int_compare:nNnTF {#1}<{10000}
      {\vskip\ulColumnBottomGlue\relax\penalty#1\relax
        % Keep the inverse separate: glue arithmetic can discard lower orders
        % (and subtraction from zero does not negate a higher-order stretch).
        \vskip-\l__ulsbs_layout_bottom_skip\relax
        \vskip\l__ulsbs_layout_previous_skip}
      {\vskip\l__ulsbs_layout_previous_skip}
  }

% No arguments; emit bottom glue and an unconditional column/page break.
\cs_new_protected:Npn \__ulsbs_layout_break:
  {\par\vskip\ulColumnBottomGlue\relax\penalty-10000\relax}

% #1: box register; #2: first split height; #3: additional split count;
% #4: first additional height (later ones use textheight). Return TF according
% to whether the copied list drains; preserve the original box and report the
% result across the private probe group via the global fits boolean.
\prg_new_protected_conditional:Npnn \__ulsbs_layout_fits:Nnnn #1#2#3#4 {TF}
  {\group_begin:
   \vbadness=10000\vfuzz=\maxdimen
   \splittopskip=0pt\splitmaxdepth=\maxdepth
   \box_set_eq:NN \l__ulsbs_layout_probe_box #1
   \tex_setbox:D \l__ulsbs_layout_split_box=
     \tex_vsplit:D \l__ulsbs_layout_probe_box to\dimexpr#2\relax
   \int_step_inline:nn {#3}
     {\box_if_empty:NF \l__ulsbs_layout_probe_box
       {\tex_setbox:D \l__ulsbs_layout_split_box=
          \tex_vsplit:D \l__ulsbs_layout_probe_box to\dimexpr
            \int_compare:nNnTF {##1}={1}{#4}{\textheight}\relax}}
   \box_if_empty:NTF \l__ulsbs_layout_probe_box
     {\bool_gset_true:N \g__ulsbs_layout_fits_bool}
     {\bool_gset_false:N \g__ulsbs_layout_fits_bool}
   \group_end:
   \bool_if:NTF \g__ulsbs_layout_fits_bool {\prg_return_true:}{\prg_return_false:}}

% #1: owned box register, consumed by unboxing. Probe placement, add intersong
% glue, tag the real tail for Lua continuation tracking, then feed actual nodes
% to LaTeX's page builder. Deferred record/write nodes run only if shipped.
\cs_new_protected:Npn \__ulsbs_layout_submit:N #1
  {
    \par
    % Let pending natural overflow reach the page builder before fit lookahead.
    \dim_compare:nNnT {\pagetotal}>{\@colroom+\pageshrink}
      {\penalty\ulSongPenalty\relax}
    \dim_set:Nn \l__ulsbs_layout_available_dim {\@colroom-\pagetotal+\pageshrink}
    \ifinner\else
      \int_compare:nNnT {\g__ulsbs_layout_columns_int}>{0}
