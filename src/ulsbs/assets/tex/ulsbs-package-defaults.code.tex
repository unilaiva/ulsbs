% Load-time settings: define with \providecommand or \newcommand in the main
% document before loading ulsbs (the songbook class also supplies options here).
% Geometry consumes these values during import; post-load \renewcommand does
% not reconfigure paper or margins. Chord-line values are read by the later
% ulChordLineParams hook, so those macros may also be renewed after loading.

% Set up options for the geometry package: paper size and margins.
% \ulTopMargin must be at least 6.35mm (=1/4"), as that is the limit of some
% printers. \ulSideMargin and \ulBottomMargin must be somewhat larger, at least
% 7.7mm, to accommodate protrusion and still be outside the 6.35mm limit.
%
% \ulBindingOffset is the extra amount of empty left in the inner sides of pages,
% and should be set to the amount the binding takes.
%
%
% Set \ulBindingOffset to used binding system's margin (from paper edge to
% beginning of the hole) + hole's width.
%
% For the default 2:1 wire binding machine with a square hole (e.g. RM SUPU
% CW360T), the hole's width is 3,5 mm and margin is adjustable. The recommended
% margin sizes for such a machine, when using 80 gsm paper, are:
%
%   SHEET COUNT  -  WIRE DIAMETER  -  MARGIN SIZE
%       2-20     -   6 mm (1/4")   -   2 mm
%      21-40     -   8 mm (5/16")  -   2 mm
%      41-55     -  10 mm (3/8")   -   2 mm
%      56-65     -  12 mm (1/2")   -   2,5 mm
%      66-100    -  14 mm (9/16")  -   2,5 mm
%     101-115    -  16 mm (5/8")   -   2,5 mm
%     116-135    -  19 mm (3/4")   -   3,5 mm
%     136-180    -  22 mm (7/8")   -   3,5 mm
%     181-210    -  25 mm (1")     -   3,5 mm
%     211-240    -  28 mm (1"1/8)  -   4,5 mm
%     241-280    -  32 mm (1"1/4)  -   4,5 mm
%     281-310    -  35 mm (1"3/8)  -   4,5 mm
%     311-340    -  38 mm (1"1/2)  -   5 mm
%     341-410    -  45 mm (1"3/4)  -   5 mm
%     411-599    -  51 mm (2")     -   5 mm
%
% NOTE: sheet count = page count / 2. Remember to add the hole's width! Take
%       into account, if you have thicker cover pages.
%
% NOTE: If in the logs fancyhdr warns that \headheight, \headsep, or \footskip
%       is too low, set \ulHeadHeight, \ulHeadSep or \ulFootSkip to the correct
%       value with \newcommand before loading this package.

% === Paper and class-size defaults ===

\providecommand*{\ulPaperSize}{a5paper}
\providecommand*{\ulBindingOffset}{8mm} % 8 mm for 2:1 wire binding, 32 mm wires

\ifcase\@ptsize\relax % 10 pt:
  % This is the default, set further down
\or % 11 pt:
  \providecommand*{\ulFootSkip}{4.5pt}
  \providecommand*{\ulHeadHeight}{6mm}
  \providecommand*{\ulHeadSep}{7mm}
  \providecommand*{\ulChordLineBaselineSkip}{11pt}%
  \providecommand*{\ulChordLineLineskipLimit}{.8pt}%
  \providecommand*{\ulChordLineLineskip}{.8pt}%
\or % 12 pt:
  \providecommand*{\ulFootSkip}{5.5pt}
  \providecommand*{\ulHeadHeight}{7mm}
  \providecommand*{\ulHeadSep}{8mm}
  \providecommand*{\ulChordLineBaselineSkip}{12pt}%
  \providecommand*{\ulChordLineLineskipLimit}{1pt}%
  \providecommand*{\ulChordLineLineskip}{1pt}%
\fi

\ifstrequal{\ulPaperSize}{a4paper}{%
  \providecommand*{\ulSideMargin}{12mm}
  \providecommand*{\ulTopMargin}{9mm}
  \providecommand*{\ulBottomMargin}{12mm}
}{}

% === Fallback defaults for A5 paper and 10pt ===

\providecommand*{\ulSideMargin}{9mm}
\providecommand*{\ulTopMargin}{6.35mm}
\providecommand*{\ulBottomMargin}{9mm}
\providecommand*{\ulFootSkip}{3.60004pt}% default
\providecommand*{\ulHeadHeight}{5mm}
\providecommand*{\ulHeadSep}{4mm}
\providecommand*{\ulChordLineBaselineSkip}{10.6pt}
\providecommand*{\ulChordLineLineskipLimit}{.5pt}
\providecommand*{\ulChordLineLineskip}{.5pt}
