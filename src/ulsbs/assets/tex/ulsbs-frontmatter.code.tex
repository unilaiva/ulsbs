% Creates a cover page here. Ought to be the first command used after
% \begin{document}.
% There is one mandatory parameter, which is the cover PDF's file name without
% path. It should reside in any of the project's include folders set up by
% the ulsbs compiler. If the filename is left empty, a fallback page with only
% the songbook name is generated in its stead.
%
% Optionally set icons in the document preamble with ulSetPermanentCoverIcons
% and ulSetVariableCoverIcons. For generated variants, the unstarred command
% replaces variable icons with type-specific ones; the star clears those slots.
% Permanent slots are unchanged. The ordinary chorded type keeps variable icons.
\NewDocumentCommand{\ulCoverPage}{s m}{%
  \IfBooleanTF{#1}{% starred: clear variant icons for the active book type
    \ifulLyricOnly
      \ulSetVariableCoverIcons%
    \else\ifulCharangoBook
      \ulSetVariableCoverIcons%
    \else\ifulBassclefBook
      \ulSetVariableCoverIcons%
    \fi\fi\fi
  }{% unstarred: set variant icons for the active book type
    \ifulLyricOnly
      \ulSetVariableCoverIcons[\ulLyricsOnlyTagIcon]%
    \else\ifulCharangoBook
      \ulSetVariableCoverIcons[\ulChordsTagIcon][\ulTrebleclefTagIcon][\ulCharangoTagIcon]%
    \else\ifulBassclefBook
      \ulSetVariableCoverIcons[\ulBassclefTagIcon]%
    \fi\fi\fi
  }%
  \IfBlankTF{#2}{%
    % fallback page
    \clearpage
    \thispagestyle{empty}%
    \null
    \begin{center}
      \vfill
      {\Huge \ulMainBookTitle}%
      \vfill
    \end{center}
    \ulsbs@ulInsertCoverIcons
    \clearpage
  }{%
    % normal PDF
    \includepdf[pages={1},pagecommand=\ulsbs@ulInsertCoverIcons]{#2}%
  }%
}


% No arguments: typeset the imprint, usually on the verso after the cover.
% Reads the loader's book-title/version/website/variant defaults; presentation
% changes are grouped. This command does not itself force a page break.
\newcommand{\ulImprintPage}{
  { % must be contained within a block for the settings to not bleed
    \thispagestyle{empty}
    \topskip0pt
    \normalsize
    \vspace*{11.2ex} % align with "Contents" heading on the next page; BAD for any changes
    \begin{center}
      % Main title
      {\Huge \ulMainBookTitle}
      % Motto
      \ifdefempty{\ulBookMotto}{}{% \ulBookMotto exists
        \par\textit{\normalsize\ulBookMotto}
      }%
      % Subtitle (e.g. Part I)
      \ifdefempty{\ulSubBookTitle}{}{% \ulSubBookTitle exists
        \par\vspace{2ex}
        {\Large \textbf{\ulSubBookTitle}}
        \ifdefempty{\ulSubSubBookTitle}{}{% \ulSubSubBookTitle exists
          \par\vspace{0.618034ex}%
          {\normalsize \ulSubSubBookTitle}%
        }%
      }%
      \par\vspace*{\fill}
      % Variant
      {\large \ulBookVariantPreText}
      \par\vspace{0.618034ex}
      {\Large%
        \par\large\emph{%
          \ifulLyricOnly%
            \ulBookLyricVariantText%
          \else%
            \ifulCharangoBook%
              \ulBookCharangoVariantText%
            \else%
              \ifulBassclefBook%
                \ulBookBassclefVariantText%
              \else%
                \ulBookDefaultVariantText%
              \fi%
            \fi%
          \fi}%
        \ulVariantIconsTextSize%
      }
      \par\vspace{2ex}
      % Version
      {\large \ulBookVersionPreText}
      \par\vspace{0.618034ex}
      {\Large%
        %vvvv-mm-dd % Use hardcoded date date for tagged printout releases
        \ulBookVersion% current date, the default
      }
      % By
      \par\vspace*{\fill}
      \ifdefempty{\ulBookByText}{}{% \ulBookByText exists
        {\large \ulBookByText}%
      }%
      \par\vspace*{\fill}
      % Link pretext (used in selections)
      \def\ulsbs@showBookWebsiteLinkPreText{true}%
      \ifdefempty{\ulBookWebsiteLinkQrImage}{}{% not empty
        \ifdefempty\ulBookWebsiteLink{}{% not empty
          \let\ulsbs@showBookWebsiteLinkPreText\undefined%
        }%
      }%
      \ifdefined\ulBookWebsiteLinkPreText\else\let\ulsbs@showBookWebsiteLinkPreText\undefined\fi%
      \ifdefined\ulsbs@showBookWebsiteLinkPreText%
        {\ulBookWebsiteLinkPreText}\par%
      \fi
      % website QR image
      \ifdefempty{\ulBookWebsiteLinkQrImage}{}{% not empty
        \imagel[3]{\ulBookWebsiteLinkQrImage} % is already centered, so 'l' version ok
        \par
      }%
      % website textual link
      \ifdefempty{\ulBookWebsiteLink}{}{% not empty
        {\small\url{\ulBookWebsiteLink}}%
      }%
      % Add vspace once, if either \ulBookWebsiteLink or \ulBookWebsiteLinkQrImage,
      % or both, exist
      \ifdefempty{\ulBookWebsiteLinkQrImage}%
        {% qr empty
          \ifdefempty{\ulBookWebsiteLink}{}{% link not empty
            \par\vspace*{\fill}%
          }%
        }%
        {% qr not empty
          \par\vspace*{\fill}
        }%
      \ifdefempty{\ulBookCompiledByText}{}{% not empty
        {\footnotesize\ulBookCompiledByText}
        \par\vspace*{\fill}
      }%
      \ifdefempty{\ulImprintPageFootnote}{}{% not empty
        {\scriptsize\ulImprintPageFootnote}
      }%
    \end{center}
    \vspace{-1.85ex}% workaround to align page bottom with other pages
  }
}

% No arguments: emit contents with the TOC page style and localized page label.
% Writes TOC material, toggles protrusion off/on, then starts a new page and
% restores the ulsbs page style (not an arbitrary previous style).
\newcommand{\ulToc}{
  \pagestyle{ulsbstoc}% use alternate header style for TOC
  % to have totally flat right edge of the dots in TOC (microtype package):
  \microtypesetup{protrusion=false} % To have a flat right side in TOC
  \tableofcontents % The TOC
  \microtypesetup{protrusion=true}% back to default (microtype package)
  % Remove the overlapping page number in the bottom of the first TOC page:
  \addtocontents{toc}{\protect\thispagestyle{empty}}
  % Adds text "Page" on top of the TOC:
  \ulIfLanguage{portuguese}{%
    \addtocontents{toc}{~\hfill\textbf{Página}\par}
  }{%
    \ulIfLanguage{finnish}{%
      \addtocontents{toc}{~\hfill\textbf{Sivu}\par}
    }{
      \addtocontents{toc}{~\hfill\textbf{Page}\par}
    }
  }
  \newpage\pagestyle{ulsbs}% return to the default header style
}

\def\ulsbs@currentchaptercolor{white}

% Outputs the chapter symbol defined above in color; the mandatory parameter
% is the color name.
\newcommand\ulsbs@coloredchaptersymbol[1]{%
  {\textcolor{#1}\ulChapterSymbol}%
}

% Define a main book chapter; index chapters instead use songchapter.
%
% Usage: \ulMainChapter*[short title]{long title}{chapter color}[image filename]
%
% #1: optional star forwarded to chapter for an unnumbered chapter.
% #2: optional short title for TOC/headers; omitted means use the long title.
% #3: mandatory long title for the chapter front page.
% #4: mandatory color name, globally saved for chapter corner marks; also used
%     for the chapter symbol when no image is supplied.
% #5: optional image filename, replacing the symbol (there is no second star).
% Omit optional brackets entirely when the argument is absent; [] is supplied
% empty content, not a request for a default.
%
% Examples:
%   \ulMainChapter{Good songs}{red}
%   \ulMainChapter[Nice chapter]{Very very nice chapter}{blue}
%   \ulMainChapter*{Bad songs}{gray}[bad_songs_image.png]
%
% Effects: sets title/image macros in caller scope, calls chapter, resets dashed
% corner marks, optionally emits a chapter-front image/symbol, and clears page.
\NewDocumentCommand{\ulMainChapter}{s o m m o}{% star, [short title], {long title}, {color}, [image]
  \IfValueTF{#2}{% short title is given
    \def\ulsbs@shorttitletext{#2}%
  }{ % short title is not given, use the long one as the short title
    \def\ulsbs@shorttitletext{#3}%
  }%
  \def\ulsbs@longtitletext{#3}%
  \gdef\ulsbs@currentchaptercolor{#4}%
  \IfValueTF{#5}{% yes, there is image given
    \def\ulsbs@smallchapterimage{%
      \includegraphics[height=1em,keepaspectratio=true]{#5}%
    }%
    \def\ulsbs@largechapterimage{%
      \vspace{-1em}%
      \imagecc[1]{#5}%
    }%
  }{% no image given, use symbol instead
    \def\ulsbs@smallchapterimage{\ulsbs@coloredchaptersymbol{#4}}%
    \def\ulsbs@largechapterimage{%
      \vfill\vspace{-1em}%
      \begin{center}%
        \fontsize{108pt}{128pt}\ulsbs@coloredchaptersymbol{#4}%
      \end{center}%
      \vfill%
    }%
  }%
  \def\ulsbs@shorttitlefull{%
    % Include the image in the chapter title string, if the image is given and
    % the feature is turned on.
    \texorpdfstring{% pdf indexes can't contain images (and \MakeUppercase must be protected)
      \ifulUppercaseShortChapterTitle\protect\MakeUppercase{\expandafter\ulsbs@shorttitletext}\else\ulsbs@shorttitletext\fi%
      \ifulShowImageInChapterName\ifdefined\ulsbs@smallchapterimage\hspace{1em}\ulsbs@smallchapterimage\fi\fi%
    }{\protect\MakeUppercase{\expandafter\ulsbs@shorttitletext}}%
  }
  \def\ulsbs@longtitlefull{%
    \ifulUppercaseLongChapterTitle\protect\MakeUppercase{\expandafter\ulsbs@longtitletext}\else\ulsbs@longtitletext\fi%
  }
  \IfBooleanTF{#1}{% \ulMainChapter with star, call \chapter with star also
    \chapter*[\ulsbs@shorttitlefull]{\ulsbs@longtitlefull}%
  }{% \ulMainChapter without star, call \chapter without it also
    \chapter[\ulsbs@shorttitlefull]{\ulsbs@longtitlefull}%
  }
  % Make chapter corner marks to be non-dashed:
  \ulsbs@chapcornermarkdashedfalse
  % The rest setups the chapter's front page
  \thispagestyle{empty}% to omit the page number in the bottom of the page
  \ifulShowImageOnChapterFrontPage\ifdefined\ulsbs@largechapterimage\vspace{-1em}\ulsbs@largechapterimage\fi\fi%
  \clearpage%
}
