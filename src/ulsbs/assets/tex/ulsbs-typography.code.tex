% Load: ulsbs.sty, after PDF color-profile setup, before colors and song fonts.
% Requires: microtype, fontspec, polyglossia, newunicodechar and graphicx;
% the preliminary main language has already been selected. Noto fonts must be
% installed. Persian font aliases precede Persian language registration.
% State: inherits @ as a letter, without a wrapper group or catcode changes.
% Declares font families, changes document family defaults and microtype sets,
% registers Finnish/Persian and Unicode mappings, and allocates a symbol box.
% Font selectors and symbol commands take no arguments; their effects follow
% ordinary TeX scoping. symdim/symhalfdim isolate their font changes internally.

\ProvidesFile{ulsbs-typography.code.tex}[Fonts, languages and symbols]

% No arguments: emit the degree character used to size diminished symbols.
\newcommand*{\ulsbs@degreechar}{\char"00B0\relax}
\newsavebox{\ulsbs@dimbox}

% No arguments: diminished symbol, a small circle in the chord font.
\DeclareRobustCommand{\symdim}{%
  {\ulNotoSansFont\mdseries\upshape\ulsbs@degreechar}%
}

% No arguments: half-diminished symbol, scaled/raised ø matching ° dimensions.
% Uses graphicx's \scalebox and a locally measured scratch box.
\DeclareRobustCommand{\symhalfdim}{%
  \begingroup
  \ulNotoSansFont\mdseries\upshape
  \sbox{\ulsbs@dimbox}{\ulsbs@degreechar}%
  \leavevmode
  \raisebox{.40em}[\ht\ulsbs@dimbox][\dp\ulsbs@dimbox]{%
    \makebox[\wd\ulsbs@dimbox][c]{%
      \scalebox{.65}{\char"00F8\relax}%
    }%
  }%
  \endgroup
}
