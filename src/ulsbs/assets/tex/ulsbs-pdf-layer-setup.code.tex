% No arguments: read effective original \ulSongKey and globally cache tonic/mode.
% Unsupported nonempty keys warn; the saved tuple is cleared before parsing.
% The cache is independent of source transposition and the body's local group.
\cs_new_protected:Npn \__ulsbs_pdf_key_save:
 {
  \tl_gclear:N \g__ulsbs_pdf_transpose_key_tl
  \tl_set:Nx \l__ulsbs_pdf_key_tl {\tl_to_str:V \ulSongKey}
  \regex_replace_all:nnN {\{\}} {} \l__ulsbs_pdf_key_tl
  \regex_replace_all:nnN {\s+} {} \l__ulsbs_pdf_key_tl
  \tl_trim_spaces:N \l__ulsbs_pdf_key_tl
  \regex_extract_once:nVNTF {\A([A-G])(\#|\&|\\shrp|\\flt)?(m?)\Z}
    \l__ulsbs_pdf_key_tl \l__ulsbs_pdf_key_seq
   {
    \int_set:Nn \l__ulsbs_pdf_key_letter_int
      {\str_case:en {\seq_item:Nn \l__ulsbs_pdf_key_seq {2}}
        {{C}{0}{D}{1}{E}{2}{F}{3}{G}{4}{A}{5}{B}{6}}}
    \tl_set:Nx \l__ulsbs_pdf_key_tl {\seq_item:Nn \l__ulsbs_pdf_key_seq {3}}
    \regex_match:nVTF {\#|\\shrp} \l__ulsbs_pdf_key_tl
      {\int_set:Nn \l__ulsbs_pdf_accidental_int {1}}
      {\int_set:Nn \l__ulsbs_pdf_accidental_int
        {\tl_if_empty:NTF \l__ulsbs_pdf_key_tl {0}{-1}}}
    \int_set:Nn \l__ulsbs_pdf_key_pitch_int
      {\int_mod:nn {12+
        \clist_item:Nn \c__ulsbs_pdf_natural_clist {\l__ulsbs_pdf_key_letter_int+1}+
        \l__ulsbs_pdf_accidental_int}{12}}
    \int_set:Nn \l__ulsbs_pdf_key_minor_int
      {\str_if_eq:eeTF {\seq_item:Nn \l__ulsbs_pdf_key_seq {4}}{m}{1}{0}}
    \tl_gset:Nx \g__ulsbs_pdf_transpose_key_tl
      {{\int_use:N \l__ulsbs_pdf_key_letter_int}
       {\int_use:N \l__ulsbs_pdf_key_pitch_int}
       {\int_use:N \l__ulsbs_pdf_key_minor_int}}
   }
   {\tl_if_empty:NF \l__ulsbs_pdf_key_tl
     {\PackageWarning{ulsbs}{Song~\tl_to_str:V\ulSongTitle :~unsupported~key=\tl_to_str:V\ulSongKey;
       ~using~current~sharp/flat~preference~for~transposition}}}
 }

% No arguments: append pending safe notation as original-footprint alternatives
% to the output token list, then clear the pending chunk.
\cs_new_protected:Npn \ulsbs@pdfchunkflush
 {
  \tl_if_empty:NF \l__ulsbs_pdf_chunk_tl
   {\tl_put_right:Nx \l__ulsbs_pdf_output_tl
     {\exp_not:N\ulsbs@pdfvisual{\exp_not:V\l__ulsbs_pdf_chunk_tl}{\exp_not:N\@firstofone}{1}}
    \tl_clear:N \l__ulsbs_pdf_chunk_tl}
 }

% #1 = token/group: append without re-executing notation-side effects.
\cs_new_protected:Npn \ulsbs@pdfchunkadd #1 {\tl_put_right:Nn \l__ulsbs_pdf_chunk_tl {#1}}

% #1 = shared content: flush notation before appending it unchanged.
\cs_new_protected:Npn \ulsbs@pdfsharedadd #1
 {\ulsbs@pdfchunkflush\tl_put_right:Nn \l__ulsbs_pdf_output_tl {#1}}
\cs_generate_variant:Nn \regex_match:nnTF {nV}

% #1 = group contents: accept safe literals or fall back for arbitrary commands.
\cs_new_protected:Npn \ulsbs@pdfgroup #1
 {
  \tl_set:Nx \l_tmpa_tl {\tl_to_str:n {#1}}
  % The chord scanner preserves braced groups without inspecting them.
  % Arbitrary nested commands are unsafe in eleven extra visual boxes.
  \regex_match:nVTF {\\|\^} \l_tmpa_tl
    {\ulsbs@pdfunknown{{#1}}}
    {\ulsbs@pdfchunkadd{{#1}}\ulsbs@pdfsplit}
 }

% #1 = chord token: route to safe notation, shared payload or explicit fallback.
\cs_new_protected:Npn \ulsbs@pdfsplittoken #1
 {
  % Test the character, not its meaning: replay rebinds active ^.
  % It must never be re-evaluated in an alternative representation.
  \str_if_eq:eeTF {\token_to_str:N #1}{\char_generate:nn {94}{12}}
   {\ulsbs@pdfsharedadd{#1}\ulsbs@pdfsplit}
   {
    \token_if_eq_meaning:NNTF #1\ulsbs@melodyreplaypayload
      {\ulsbs@pdfsplitpayload}
      {\token_if_eq_meaning:NNTF #1\ac
        {\ulsbs@pdfsplitac}
        {\token_if_eq_meaning:NNTF #1\notrans
          {\ulsbs@pdfsplitnotrans}
          {\token_if_active:NTF #1
            {\ulsbs@pdfchunkadd{#1}\ulsbs@pdfsplit}
            {\token_if_cs:NTF #1
            {\clist_if_in:nnTF
              {\shrp,\flt,\symdim,\symhalfdim,\textdegree,
               \textsuperscript,\textsubscript,\up,
               \kern,\hspace,\hphantom,\phantom,\strut,\relax,
               \ulsbs@smraiseopenparen,\ulsbs@smraisecloseparen}
              {#1}{\ulsbs@pdfchunkadd{#1}\ulsbs@pdfsplit}
              {\ulsbs@pdfunknown{#1}}}
            {\ulsbs@pdfchunkadd{#1}\ulsbs@pdfsplit}}}}}
   }
 }

% #1 = unsupported token/group, #2 = tail through the split terminator.
% Restore the complete original chord text and warn rather than replay it.
\cs_new_protected:Npn \ulsbs@pdfunknown #1#2\ulsbs@pdfsplitend
 {
  % Unknown commands may consume dimensions, optional or unbraced arguments.
  % Preserve the complete tail, rather than injecting boxes into that stream.
  \tl_set_eq:NN \l__ulsbs_pdf_output_tl \l__ulsbs_pdf_original_tl
  \tl_clear:N \l__ulsbs_pdf_chunk_tl
  \PackageWarning{ulsbs}{Song~\ulsbs@pdfsongid :~unsupported~chord~command/group~
    \tl_to_str:n {#1};~interactive~transposition~is~not~applied~to~this~chord~text}
 }

% #1 = effective chord text: reset splitter buffers, split into transposable
% chunks and unchanged shared items, then execute the resulting output.
\cs_new_protected:Npn \ulsbs@pdfchordtext #1
 {
  \tl_clear:N \l__ulsbs_pdf_chunk_tl
  \tl_clear:N \l__ulsbs_pdf_output_tl
  \tl_set:Nn \l__ulsbs_pdf_original_tl {#1}
  \ulsbs@pdfsplit #1\ulsbs@pdfsplitend
  \tl_use:N \l__ulsbs_pdf_output_tl
 }

% No arguments: peek without expanding active notation/replay commands.
\def\ulsbs@pdfsplit{\futurelet\ulsbs@pdfnext\ulsbs@pdfsplitdispatch}

% No arguments: route the peeked terminator, group, space or ordinary token.
\def\ulsbs@pdfsplitdispatch{%
  \ifx\ulsbs@pdfnext\ulsbs@pdfsplitend\let\ulsbs@pdfnextstep\ulsbs@pdfsplitfinish
  \else\ifx\ulsbs@pdfnext\bgroup\let\ulsbs@pdfnextstep\ulsbs@pdfsplitgroup
  \else\ifcat\noexpand\ulsbs@pdfnext\@sptoken\let\ulsbs@pdfnextstep\ulsbs@pdfsplitspace
  \else\let\ulsbs@pdfnextstep\ulsbs@pdfsplitstep\fi\fi\fi
  \ulsbs@pdfnextstep}

% Consume the terminator and emit the final pending chord chunk.
\def\ulsbs@pdfsplitfinish\ulsbs@pdfsplitend{\ulsbs@pdfchunkflush}

% #1 = a braced group: consume it while retaining its grouping boundaries.
\def\ulsbs@pdfsplitgroup#1{\ulsbs@pdfgroup{#1}}

% #1 = an ordinary token: pass it through the safe-token dispatcher.
\def\ulsbs@pdfsplitstep#1{\ulsbs@pdfsplittoken{#1}}

% #1/#2 = already captured melody payload arguments: preserve as shared content.
\def\ulsbs@pdfsplitpayload#1#2{\ulsbs@pdfsharedadd{\ulsbs@melodyreplaypayload{#1}{#2}}\ulsbs@pdfsplit}

% #1/#2 = alternate annotation arguments: share without extra parser executions.
\def\ulsbs@pdfsplitac#1#2{\ulsbs@pdfsharedadd{\ac{#1}{#2}}\ulsbs@pdfsplit}

% #1 = notrans contents: preserve wrapper and argument as one shared parser item.
\def\ulsbs@pdfsplitnotrans#1{\ulsbs@pdfsharedadd{\notrans{#1}}\ulsbs@pdfsplit}
